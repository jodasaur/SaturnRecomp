function(saturn_test name)
    add_executable(test_${name} ${name}.c)
    target_compile_options(test_${name} PRIVATE -UNDEBUG)
    target_link_libraries(test_${name} PRIVATE ${ARGN})
    if(name STREQUAL "sh2_fastpath_fuzz")
        add_test(NAME ${name} COMMAND test_${name} 20000)
    else()
        add_test(NAME ${name} COMMAND test_${name})
    endif()
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 120)
endfunction()
